using System.Collections;
using System.Collections.Generic;
using UnityEngine;

public class Cameracontrol : MonoBehaviour
{
    //保存向量
    private Vector3 vector;
    //关联玩家
    public GameObject player;
    // Start is called before the first frame update
    void Start()
    {
        //获取从摄像机到玩家的向量
        vector = player.transform.position-transform.position;

    }

    // Update is called once per frame
    void Update()
    {
        //执行计算，时刻更新摄像机的位置进行跟随
        transform.position = player.transform.position-vector;
    }
}

using UnityEngine;

    private GameObject bo;
    // Start is called before the first frame update
    void Start()
    {
        //获取一下游戏物体身上的音频源
        Bomb = GetComponent<AudioSource>();
        Bomb.clip = Resources.Load<AudioClip>("bomb");//通过路径获取音乐
        //文件几名好像只能是Resources

    }

    // Update is called once per frame
    void Update()
    {
        bo = GameObject.Find("Bomb(Clone)");
        if(bo != null)
        {
            //播放音效
            Invoke("Bofang", 2f);
        }
    }
    void Bofang()
    {
        //播放音效
        Bomb.PlayOneShot(Bomb.clip);
    }
}
炸弹控制
using System.Collections;
using System.Collections.Generic;
using UnityEngine;

public class Bombcontrol : MonoBehaviour
{
    //创建一个关联的变量，关联一个爆炸效果
    public GameObject EffectPer;

    // Start is called before the first frame update
    void Start()
    {
        //该方法是2秒以后执行"Boom"方法
        Invoke("Boom", 2f);
    }

    void Boom()
    {

        //创建一个游戏物体，把他实例化成爆炸效果
        GameObject effect = Instantiate(EffectPer, transform.position, transform.rotation);

        //2秒后删除爆炸效果
        Destroy(effect,2f);

        //判断是否炸到敌人
        //1.获取场景中的所有敌人（用标签的方法，获取该标签的所有物体，只要敌人都加上该标签即可获取敌人）
        GameObject[] enemys = GameObject.FindGameObjectsWithTag("Enemys");
        //遍历敌人(也可以用foreach遍历i，像书中一样）
        for(int i = 0; i < enemys.Length; i++)
        {
            GameObject enemy = enemys[i];
            if(Vector3.Distance(transform.position, enemy.transform.position) < 3f)
            {
                Destroy(enemy);
            }
        }

        //破坏炸弹本体
        Destroy(gameObject);
    }

    // Update is called once per frame
    void Update()
    {
        
    }
}

using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.SceneManagement;

public class Enemycotrol : MonoBehaviour
{
    //获取玩家
    private GameObject player;
    //僵尸音效
    private AudioSource jiangshi;

    // Start is called before the first frame update
    void Start()
    {
        //通过标签找到玩家
        player = GameObject.FindWithTag("player");
        //获取音源
        jiangshi = GetComponent<AudioSource>();
        //获取音乐
        jiangshi.clip = Resources.Load<AudioClip>("丧尸");
        jiangshi.loop = true;
        jiangshi.Play();
    }

    // Update is called once per frame
    void Update()
    {
        //获取与玩家的距离
        float distance = Vector3.Distance(player.transform.position,transform.position);
        //判断与玩家的距离
        if (distance < 5f)
        {
            //追踪玩家
            //看向玩家
            transform.LookAt(player.transform);
            //向玩家移动
            transform.Translate(Vector3.forward * 4f * Time.deltaTime);
        }
        //如果与玩家距离小于2米（两者中心点为两米说明两者差不多挨着）
        if(distance < 1.5f)
        {
            //Debug.Log("游戏失败！");
            SceneManager.LoadScene(3, LoadSceneMode.Single);
        }
    }
}

using System.Collections;
using System.Collections.Generic;
using System.Diagnostics;
using UnityEngine;

public class Playercontrol : MonoBehaviour
{
    //获取摇杆
    public FloatingJoystick floatingjoystick;


    //获取音频源
    private AudioSource player;
    //声明一个变量储存物体当前的位置
    private Vector3 now;

    //制作人物放炸弹的功能
    //关联一下炸弹的预制体
    public GameObject BombPre;
    //炸弹的CD
    private float CD = 2;//先设置为2是为了保证第一发炸弹可随时放）


    void Start()
    {
        //获取一下游戏物体身上的音频源
        player = GetComponent<AudioSource>();
        player.clip = Resources.Load<AudioClip>("脚步");//通过路径获取音乐
                       //文件几名好像只能是Resources
    }


    void Update()
    {
        //获取一下当前位置（一定要写在Update中，并且写在控制移动的代码之前）
        now = gameObject.transform.position;
        //控制移动(用键盘）
        //float horizontal = Input.GetAxis("Horizontal");
        //float vertical = Input.GetAxis("Vertical");

        //改为摇杆控制
        float horizontal = floatingjoystick.input.x;
        float vertical = floatingjoystick.input.y;

        Vector3 dir = new Vector3(horizontal, 0, vertical);
        if(dir != Vector3.zero)
        {
            transform.rotation = Quaternion.LookRotation(dir);
            transform.Translate(Vector3.forward * 5 * Time.deltaTime);
        }

        //判断物体是否不移动
        if (now != gameObject.transform.position)
        {
            player.volume = 0.5f;
            //UnityEngine.Debug.Log("移动了");
            if(player.isPlaying)
            {
                
            }
            else
            {
                player.Play();
            }
        }
        else
        {
            player.Stop();
        }

        //制作一个CD的计时器
        CD += Time.deltaTime;

        //当按下空格时
        if (Input.GetButtonDown("Jump"))
        {
            //判断一下是否满足CD
            if (CD > 2)
            {
                //重置CD
                CD = 0;
                //释放炸弹(实例化出来一个炸弹）
                Instantiate(BombPre, transform.position, transform.rotation);
            }
        }
    }
    public void Put()
    {
        //判断一下是否满足CD
        if (CD > 2)
        {
            //重置CD
            CD = 0;
            //释放炸弹(实例化出来一个炸弹）
            Instantiate(BombPre, transform.position, transform.rotation);
        }
    }
}
