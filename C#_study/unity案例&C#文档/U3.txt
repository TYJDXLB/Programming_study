音频
using JetBrains.Annotations;
using System.Collections;
using System.Collections.Generic;
using Unity.VisualScripting;
using UnityEngine;

public class audio : MonoBehaviour
{
    //声明两个Audio Clip（音频）
    public AudioClip music;
    public AudioClip se;

    //声明一个组件类型的变量
    private AudioSource player;

    // Start is called before the first frame update
    void Start()
    {
        //获取一下组件
        player = GetComponent<AudioSource>();
        //设定播放的音频片段
        player.clip = music;
        //循环播放      (true为循环）
        player.loop = true;
        //音量
        player.volume = 0.1f;
        //播放
        player.Play();
        //停止
        player.Stop();
        //暂停
        player.Pause();
        //非暂停
        player.UnPause();


    }

    // Update is called once per frame
    void Update()
    {
        //做一个按下空格就会播放，再按就会停止
        if (Input.GetKeyUp(KeyCode.Space))
        {
            //判断一下是否正在播放
            if(player.isPlaying)
            {
                player.Stop();
            }
            else
            {
                player.Play();
            }
        }

        //鼠标按一下就播放一次
        if(Input.GetMouseButtonDown(0))
        {
            player.PlayOneShot(se);     //括号内写音效名
        }
    }
}

using System.Collections;
using System.Collections.Generic;
using UnityEngine;

public class player : MonoBehaviour
{
    //反向动力学
    //先拿到球体的transform组件
    public Transform target;

    //声明变量
    private Animator animator;
    // Start is called before the first frame update
    void Start()
    {
        //获取组件
        animator = GetComponent<Animator>();
    }

    // Update is called once per frame
    void Update()
    {
        //按F拾东西
        if(Input.GetKeyDown(KeyCode.F))
        {
            //触发pickup参数
            GetComponent<Animator>().SetTrigger("pick up");
        }
        //移动
        //水平轴
        float horizontal = Input.GetAxis("Horizontal");
        //竖直轴
        float vertical = Input.GetAxis("Vertical");
        //设置向量
        Vector3 dir = new Vector3(horizontal, 0, vertical);
        //当用户按下方向键
        if(dir != Vector3.zero)     //当方向向量不是0向量，是就是按下了方向键
        {
            //使人物面相向量      Quaternion是四元数的类     LookRotation是看向向量的方法
            transform.rotation = Quaternion.LookRotation(dir);
            //播放跑步动画
            GetComponent<Animator>().SetBool("isrun", true);//前一个参数是要修改的名称，后一个是要修改为什么状态
            //朝前方移动
            transform.Translate(Vector3.forward * 2 * Time.deltaTime);
        }
        else
        {
            //播放站立动画
            GetComponent<Animator>().SetBool("isrun", false);
        }
        //随时获得test参数数值，并打印出来
        //Debug.Log(animator.GetFloat("test"));
    }
    void leftfoot()
    {
        Debug.Log("左脚");
    }

    void rightfoot()
    {
        Debug.Log("右脚");
    }

    //IK相关的都写到这个方法中
    private void OnAnimatorIK(int layerIndex)//参数为动画控制的层数
    {
        //设置头部IK
        animator.SetLookAtWeight(1);//设置头部IK权重
        animator.SetLookAtPosition(target.position);//看向 物体的位置
        //设置右手IK（位置权重）
        animator.SetIKPositionWeight(AvatarIKGoal.RightHand,1);
        //SetIKPositionWeight（）是设置某个部位的IK（除了头）权重、AvatarIKGoal是一个枚举，枚举出要设置的位置，第二个参数是权重
        //旋转权重
        animator.SetIKRotationWeight(AvatarIKGoal.RightHand, 1);
        //设置右手IK位置
        animator.SetIKPosition(AvatarIKGoal.RightHand,target.position);//右手IK的指向物体
        //设置右手IK旋转
        animator.SetIKRotation(AvatarIKGoal.RightHand,target.rotation);
    }

}

using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.AI;//需要该命名空间

public class player2 : MonoBehaviour
{
    //声明一个导航组件的变量
    private NavMeshAgent agent;
    // Start is called before the first frame update
    void Start()
    {
        //获取代理组件
        agent = GetComponent<NavMeshAgent>();
    }

    // Update is called once per frame
    void Update()
    {
        //如果按下鼠标左键
        if (Input.GetMouseButtonDown(0))
        {
            //获取点击位置（用射线）
            Ray ray = Camera.main.ScreenPointToRay(Input.mousePosition);//从摄像机发射一条射线,括号内是鼠标位置
            //声明一个碰撞信息类
            RaycastHit hit;
            //碰撞检测
            if(Physics.Raycast(ray, out hit))
            {
                //点击位置
                Vector3 point = hit.point;
                //设置该位置为导航目标点
                agent.SetDestination(point);
            }

        }
    }
}

            if(res == true)
            {
                Debug.Log(hit.point);
                transform.position = hit.point;
            }
            //多检测
            RaycastHit[] hits = Physics.RaycastAll(ray, 100, 1 << 10);//100是距离,"1<<10"是只检测第十图层的物体。
        }
    }
