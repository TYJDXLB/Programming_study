using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;

namespace lianxi13
{
    internal class Person
    {
        private string name;
        //常规属性
        public string Name
        {
            get
            {
                if (name == null || name == "")
                {
                    return "当前名字为空";
                }
                else
                {
                    return name;
                }
            }
            set { name = value; }
        }

        //自动属性
        public int Age { get; set; }


        
    }
}
