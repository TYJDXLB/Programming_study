using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;

namespace OOPtest
{
    internal class LuoTianYi
    {
        private string heroName;        //名称
        private string heroInfo;        //简介
        private int attack;             //攻击
        private int defense;            //防御
        private int speed;              //速度
        private string nickName;        //昵称

        public LuoTianYi()
        { }

        public void Hellow()
        {
            Console.WriteLine("姓名：" + this.heroName);
            Console.WriteLine("介绍：" + this.heroInfo);
            Console.WriteLine("攻击：" + this.attack);
            Console.WriteLine("防御：" + this.defense);
            Console.WriteLine("速度：" + this.speed);
            Console.WriteLine("昵称：" + this.nickName);
        }

        public LuoTianYi(string nickName)
        {
            this.heroName = "洛天依";
            this.heroInfo = "虚拟歌姬洛天依";
            this.attack = 64;
            this.defense = 77;
            this.speed = 88;
            this.nickName = nickName;
        }

        public string HeroName
        {
            get { return heroName; }
            set { heroName = value; }
        }
        public string HeroInfo
        {
            get { return heroInfo; }
            set { heroInfo = value; }
        }
        public int Attack
        {
            get { return attack; }
            set { attack = value; }
        }
        public int Defense
        {
            get { return defense; }
            set { defense = value; }
        }
        public int Speed
        {
            get { return speed; }
            set { speed = value; }
        }
        public string NickName
        {
            get { return nickName; }
            set { nickName = value; }
        }

        public void GanWuNv()
        {
            Console.WriteLine("干物女");

        public void PiaoLiangMianDui()

            Console.WriteLine("漂亮面对");
        }

}
