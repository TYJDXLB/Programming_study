Car
using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;

namespace lianxi12
{
    /// <summary>
    /// 汽车的父类.
    /// </summary>
    internal class Car
    {
        private string brand;       //建一个字段
        public string Brand         //用一个属性封装
        {
            get { return brand; }
            set { brand = value; }
        }
        public Car(string brand)    //建一个构造方法，用于初始化这个父类
        {
            this.brand = brand;
        }
        public void Run()           //一个普通方法
        {
            Console.WriteLine("{0}品牌的车在奔跑。。。",brand);
        }



    }
}
