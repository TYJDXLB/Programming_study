XuZ
using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;

namespace lianxi14
{
    internal class XuZ:XufangfaF
    {
        public void xuzi()//子类自己的方法
        {
            Console.WriteLine("子类中自己的方法");
        }

        public override void xufu()//重写父类的方法
        {
            base.xufu();
            Console.WriteLine("虚方法子类");
        }
        
    }
}
