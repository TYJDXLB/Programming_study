using System;
using System.Collections;
using System.Collections.Generic;
using System.Threading;
using UnityEngine;
using UnityEngine.SceneManagement;  //要做场景操作，要先导入这个命名空间

public class async : MonoBehaviour
{
    AsyncOperation operation;       //因为SceneManager.LoadSceneAsync(2);有个返回值，所以最开始声明一下
    
    void Start()
    {
        StartCoroutine(LoadScene());        //算是开启了一个协程
    }

    //协程方法来异步加载场景（异步比同步块）
    IEnumerator LoadScene()         //返回值固定是IEnumerator千万别写错成IEnumerable！！！！！
    {
        operation = SceneManager.LoadSceneAsync(2);     //接受一下返回值
        //不允许场景自动跳转
        operation.allowSceneActivation = false;     //true为自动跳转，默认为true

        yield return operation;

    }

    //做一个计时器，让它在5秒后再跳转
    float timer = 0;

    void Update()
    {
        //获取异步加载进度，范围就是0-0.9
        Debug.Log(operation.progress);

        timer +=Time.deltaTime;
        if(timer > 5)             //括号内输入：Input.GetKey(KeyCode.W)就是按W键跳转
        {
            operation.allowSceneActivation = true;
        }
    }
}

                    break;
