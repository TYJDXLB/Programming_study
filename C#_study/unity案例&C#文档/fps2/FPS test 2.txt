using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.UI;

public class change : MonoBehaviour
{
    private Transform ak47;
    private Transform glock;
    //判断是否为主武器
    private bool iszhu = true;

    [Header("图标设置")]
    public Image GLOCK;
    public Image AK47;
    public Text gun;

    private string name;

    [Header("检测是否武器齐全(用假的判断)")]
    public GameObject jiaAK;
    public GameObject jiaGL;

    private AudioListener listener;
    void Start()
    {
        //获取到武器物体
        ak47 = transform.FindChild("ak47");
        glock = transform.FindChild("glock");
        //调用听组件
        listener = GetComponent<AudioListener>();
    }
    void Update()
    {
        if(jiaAK.gameObject.activeSelf&&jiaGL.gameObject.activeSelf)
        {
            if (iszhu == false && (Input.GetKeyDown(KeyCode.Alpha1) || Input.GetAxis("Mouse ScrollWheel") > 0))
            {
                Debug.Log("按下了1");
                ak47.gameObject.SetActive(true);
                glock.gameObject.SetActive(false);
                iszhu = true;
                //设置UI
                AK47.gameObject.SetActive(true);
                GLOCK.gameObject.SetActive(false);
                name = "AK47";
                gun.text = name;
                //暂时“失聪”一下可以不听见多余的开抢声
                listener.enabled = false;
                Invoke("relisten", 0.25f);
            }
            if (iszhu == true && (Input.GetKeyDown(KeyCode.Alpha2) || Input.GetAxis("Mouse ScrollWheel") < 0))
            {
                Debug.Log("按下了2");
                ak47.gameObject.SetActive(false);
                glock.gameObject.SetActive(true);
                iszhu = false;
                //设置UI
                AK47.gameObject.SetActive(false);
                GLOCK.gameObject.SetActive(true);
                name = "Glock";
                gun.text = name;
                //暂时“失聪”一下可以不听见多余的开抢声
                listener.enabled = false;
                Invoke("relisten", 0.25f);
            }

        }
    }

    //恢复听力
    void relisten()
    {
        //激活听组件
        listener.enabled = true;
    }

}

using System.Collections;
using System.Collections.Generic;
using UnityEngine;

public class dun : MonoBehaviour
{
    //蹲下与站立的高度
    private float crouchh;
    private float standarh;
    //站立与蹲下的摄像机位置
    private Vector3 camerapos;
    private Vector3 cameracpose;
    //实例化相机
    private GameObject camera;
    //获取人物控制器组件
    private CharacterController cha;
    //判断是否蹲下（初始为否）
    private bool iscrouch = false;

    // Start is called before the first frame update
    void Start()
    {
        //找到相机物体（提前把摄像机的标签设置为MainCamera）
        camera = GameObject.FindGameObjectWithTag("MainCamera");
        //获取人物控制器组件
        cha = GetComponent<CharacterController>();
        //设置站立高度为人物高度
        standarh = cha.height;
        //设置蹲下高度为站立高度的一半
        crouchh = cha.height / 2f;
        //站立时相机的相对父物体的位置
        camerapos = camera.transform.localPosition;
        //蹲下时摄像机的位置相对于父物体的位置
        cameracpose = new Vector3(camerapos.x,camerapos.y/16,camerapos.z);

    }
    //蹲下函数
    void crouch()
    {
        //设置人物高度为蹲下时的高度
        cha.height = crouchh;
        //人物的中心为原本的一半
        cha.center = new Vector3(0, -0.5f, 0);
        //把摄像机的位置设置为蹲下时的位置
        camera.transform.localPosition = cameracpose;
    }
    //起立函数
    void getup()
    {
        //人物的位置设置为原本的位置
        transform.position = new Vector3(transform.position.x,transform.position.y,transform.position.z);
        //人物的中心设置为原本的中心
        cha.center = new Vector3(0, 0, 0);
        //高度设置为站立高度
        cha.height = standarh;
        //相机的位置设置为原本的位置
        camera.transform.localPosition = camerapos;
    }
    // Update is called once per frame
    void Update()
    {
        if(cha.isGrounded)//如果在地上
        {
            if (Input.GetKeyDown(KeyCode.C))//如果按下c键
            {
                if (iscrouch == false)//如果没有蹲下
                {
                    crouch();//调用蹲下函数
                    iscrouch = true;//把是否蹲下设置为是
                }
                else
                {
                    getup();//调用站立函数
                    iscrouch = false;//把是否蹲下设置为否
                }
            }
        }
        else
        {
            getup();//调用站立函数
            iscrouch = false;//把是否蹲下设置为否
        }

    }
}

using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.UI;

public class GetGun : MonoBehaviour
{
    //射线范围（捡东西的范围）
    private int range = 2;
    [Header("真正的ak")]
    public GameObject ak47;
    [Header("真正的glock")]
    public GameObject glock;
    [Header("假的（判断标准）ak47")]
    public GameObject jiaak;
    [Header("假的（判断标准）glock")]
    public GameObject jiagl;
    [Header("丢枪点")]
    public Transform cassingpoint;
    [Header("ak47模型")]
    public GameObject AK47;
    [Header("glock模型")]
    public GameObject GLOCK;

    [Header("假刀")]
    public GameObject jiaknife;

    [Header("UI设置")]
    public Image Glock;
    public Image Ak47;
    public Image Knife;

    //计时器
    private float timer = 0;    //防止丢的时候枪全丢
    private float time = 0;     //防止枪刚到手就掉
    //丢枪最小时间
    private float cd = 1f;
    private float CD = 5f;
    // Start is called before the first frame update
    void Start()
    {
        
    }

    // Update is called once per frame
    void Update()
    {
        //增加计时器
        if (timer < cd)
        {
            timer += Time.deltaTime;
        }
        //增加计时器
        if (time < CD)
        {
            time += Time.deltaTime;
        }

        //射线方向
        Vector3 dir = Camera.main.transform.forward;
        //射线检测
        RaycastHit hit;
        if(Physics.Raycast(Camera.main.transform.position, dir, out hit, range)&&Input.GetKeyDown(KeyCode.G))
        {
            //捡枪
            //如果物体的标签为ak47，并且假ak没有（表示没有ak）
            if(hit.transform.tag == "ak47"&&jiaak.gameObject.activeSelf == false)
            {
                Debug.Log("没有ak并且捡到了ak");
                //清除射线射到的物体
                Destroy(hit.collider.gameObject);
                //把真假ak设置为true，表示已经捡到了ak
                jiaak.SetActive(true);
                ak47.SetActive(true);
                //把真Glock设置为false
                glock.SetActive(false);
                //把ak的ui打开
                Ak47.gameObject.SetActive(true);
                //如果已经有glock
                if (jiagl.gameObject.activeSelf == true)
                {
                    //把glock的ui关掉
                    Glock.gameObject.SetActive(false);
                }
            }
            //如果物体的标签为glock，并且假glock没有（表示没有glock）
            if (hit.transform.tag == "glock" && jiagl.gameObject.activeSelf == false)
            {
                Debug.Log("没有glock并且捡到了glock");
                //清除射线射到的物体
                Destroy(hit.collider.gameObject);
                //把真假glock设置为true，表示已经捡到了glock
                jiagl.SetActive(true);
                glock.SetActive(true);
                //把真ak设置为false
                ak47.SetActive(false);
                //把glock的ui打开
                Glock.gameObject.SetActive(true);
                //如果已经有ak
                {
                    //把ak47的ui关掉
                    Ak47.gameObject.SetActive(false);
                }
            }
            //捡刀
            if(hit.transform.tag == "knife")
            {
                //把假刀状态设置为true
                jiaknife.SetActive(true);
                //打开刀的ui
                Knife.gameObject.SetActive(true);
                //销毁刀模型
                Destroy(hit.collider.gameObject);
            }
        }

        //丢枪
        if (Input.GetKeyDown(KeyCode.G)&&time>CD)
        {
            //清空计时器
            time = 0;
            //如果拿着ak
            if (ak47.gameObject.activeSelf == true && timer > cd)
            {
                //实例化一个ak47模型
                Instantiate(AK47, cassingpoint.position, cassingpoint.rotation);
                //把真假ak设置为false
                jiaak.SetActive(false);
                ak47.SetActive(false);
                //把ak47的ui关掉
                Ak47.gameObject.SetActive(false);
                //如果有glock
                if (jiagl.gameObject.activeSelf == true)
                {
                    //把真glock设置为true
                    glock.SetActive(true);
                    //把glock的ui打开
                    Glock.gameObject.SetActive(true);
                }
                //重置计时器
                timer = 0;
            }
            if (glock.gameObject.activeSelf == true && timer > cd)//如果拿着glock
            {
                //实例化一个glock模型
                Instantiate(GLOCK, cassingpoint.position, cassingpoint.rotation);
                //把真假glock设置为false，表示已经扔了glock
                jiagl.SetActive(false);
                glock.SetActive(false);
                //把glock的ui关掉
                Glock.gameObject.SetActive(false);
                //如果有ak
                if (jiaak.gameObject.activeSelf == true)
                {
                    //把真ak设置为true
                    ak47.SetActive(true);
                    //把ak的ui打开
                    Ak47.gameObject.SetActive(true);
                }
                //重置计时器
                timer = 0;
            }
        }
    }
}

using System.Collections;
using System.Collections.Generic;
using UnityEngine;

public class Head : MonoBehaviour
{
    //获取一下摄像机与人物的transform组件
    private Transform head;
    private Transform player;
    // Start is called before the first frame update
    void Start()
    {
        //获取一下摄像机与人物的transform组件
        head = transform;
        player = transform.parent;
        //隐藏鼠标指针
        Cursor.lockState = CursorLockMode.Locked;
    }

    // Update is called once per frame
    void Update()
    {
        Screen();
    }

    //转视角的代码
    void Screen()
    {
        //获取鼠标水平轴数值
        float mousex = Input.GetAxis("Mouse X");
        if(mousex != 0)//如果鼠标在水平方向上有移动
        {
            player.Rotate(Vector3.up * mousex * 120 * Time.deltaTime);
        }
        //获取鼠标竖直轴数值
        float mousey = Input.GetAxis("Mouse Y");
        if (mousey != 0)//如果鼠标在竖直方向上有移动
        {
            head.Rotate(Vector3.left * mousey * 120 * Time.deltaTime);
        }


        //限制上下转头角度
        if (Vector3.Angle(player.forward,head.forward)>60)
        {
            head.Rotate(Vector3.left * mousey * -120 * Time.deltaTime);
        }
    }
}

using System.Collections;
using System.Collections.Generic;
using UnityEngine;

public class PlayerC : MonoBehaviour
{
    //由于player身上没挂载动画控制器，所以动画代码都要到WeaponC里写

    //获取人物控制器组件
    private CharacterController cha;

    //速度
    private float speed;
    //是否跑步判断
    private bool isrun;
    //走路速度
    private float movespeed = 4f;
    //跑步速度
    private float runspeed = 7f;

    //下蹲速度
    private float dun = 1.5f;
    //判断是否蹲下
    private bool isdun = false;

    //跳跃高度
    private float jumphigh = 1f;
    //重力加速度
    private int g = 10;
    private Vector3 up = Vector3.zero;


    [Header("按键设置")]//设置一个标题
    [SerializeField][Tooltip("奔跑按键")]public KeyCode run;//疾跑键位设置

    [Header("声音设置")]
    [SerializeField] private AudioSource aud;
    public AudioClip walks;
    public AudioClip runs;

    void Start()
    {
        //获取一下组件
        cha = GetComponent<CharacterController>();
        aud = GetComponent<AudioSource>();

        run = KeyCode.LeftShift;
    }

    void Update()
    {
        Run();
        //isduning();
        cha.Move(up * Time.deltaTime);
        jump();
    }
    //移动函数
    void Run()
    {
        //获取水平竖直轴的数值
        float horizontal = Input.GetAxis("Horizontal");
        float vertical = Input.GetAxis("Vertical");
        //运动方向
        Vector3 move = (transform.right* horizontal + transform.forward* vertical).normalized;

        //移动
        isrun = Input.GetKey(run);
        speed = isrun?runspeed:movespeed;//与下方代码效果一致
        //if(Input.GetKey(run))
        //{
        //    cha.Move(move * runspeed * Time.deltaTime);
        //}
        //else
        //{
        //    cha.Move(move * movespeed * Time.deltaTime);
        //}
        if (cha.isGrounded&&(horizontal != 0 || vertical != 0))
        {
            ////移动声音播放
            //aud.clip = isrun ? runs : walks;//设置行走或奔跑的声音
            if (aud.isPlaying == false)
            { 
                if(isrun)
                {
                    aud.clip = runs;
                    aud.Play();
                }
                else
                {
                    aud.clip = walks;
                    aud.Play();
                }
            }
        }
        else
        {
            aud.Stop();
        }

        //蹲下
        if(Input.GetKeyDown(KeyCode.C))
        {
            if(isdun == false)
            {
                isdun = true;             
                Debug.Log("蹲下");
            }
            else
            {
                isdun = false;
                Debug.Log("正常");
            }
        }
        if(Input.GetKeyDown(KeyCode.Space)&&isdun == true)
        {
            isdun = false;
            Debug.Log("正常");
        }

        if(isdun == true)
        {
            cha.Move(move * dun * Time.deltaTime);
        }
        else
        {
            cha.Move(move * speed * Time.deltaTime);
        }
    }
    //跳跃函数
    public void jump()
    {
        bool a = cha.isGrounded;
        Debug.Log(a);
        if(cha.isGrounded)
        {
            if(Input.GetButtonDown("Jump"))
            {
                up.y = Mathf.Sqrt(2*g*jumphigh);              
            }

            if(up.y < -1)
                up.y = -1;
        }
        else
        {
            up.y -= g*Time.deltaTime;
        }
    }
    //音频播放函数
    void soundplay()
    {
        if (cha.isGrounded)
            //移动声音播放
            aud.clip = isrun ? runs : walks;//设置行走或奔跑的声音
        if (aud.isPlaying)
        {
            aud.Play();
        }
    }

}

    private float timer = 0;

    [Header("弹壳设置")]
    //抛弹壳的位置

    public GameObject ak;

    [Header("UI设置")]
    public Image firep;
    public Text bulletnow;

    public Image AK47;

    //用枚举定义全自动与半自动
    public enum ShootModle
    { 
        quan,
        ban
    }
    //定义一个枚举类型

            shootname = "全自动";
            shooting = ShootModle.quan;
            shootmodle.text = shootname;

        else if(Input.GetKeyDown(KeyCode.X) && ModeNum != 0)
        {
            ModeNum = 0;
            shootname = "半自动";
            shooting = ShootModle.ban;
            shootmodle.text = shootname;
        }
        switch (shooting)
        {
            case ShootModle.quan:
                //如果一直按住左键
                shootinput = Input.GetMouseButton(0);
                //设置为全自动射速
                firerate = 0.1f;
                break;
            case ShootModle.ban:
                //如果一下一下按左键
                shootinput = Input.GetMouseButtonDown(0);
                //设置为半自动射速

                break;

            firelight.enabled = false;

            timer += Time.deltaTime;
        }

        //获取当前动画状态(获取第0层的动画状态）
        AnimatorStateInfo info = ani.GetCurrentAnimatorStateInfo(0);
        //两种换子弹动画
        if(info.IsName("reload_ammo_left@assault_rifle_01")|| info.IsName("reload_out_of_ammo@assault_rifle_01"))
        {
            isreload = true;
        }
        else
        {
            isreload = false;
        }
        //换子弹更新UI
        if(Input.GetKeyDown(KeyCode.R)&& nowbulletC< bulletC&& bullectall>0)
        {
            renewb();
            updateui();

        //瞄准
        doaiming();

        iswalk();

    //开火函数
    void Fire()
    {
        //射速控制如果,计时器时间小于射速时间,子弹数为0,换子弹,跑步,挥刀,刺刀,则跳出该方法（不执行该方法）
        if(timer < firerate||nowbulletC <= 0||isreload|| Input.GetKey(KeyCode.LeftShift)||ishui==true||isci==true) return;

        //射击方向（向前）
        Vector3 shootdir = shooterpoint.forward;
        //碰撞检测（一定别忘了写）储存射线击中的信息
        RaycastHit hit;
        //Physics.Raycast()中的四个参数分别为：起点，方向，out hit，距离
        if (Physics.Raycast(shooterpoint.position, shootdir,out hit,range))
        {
            //防止后退时射击会莫名其妙的打在自己身上，给人物添加一个player标签
            if(hit.transform.tag != "player")
            {
                Debug.Log("打到了" + hit.transform.name);

        float x = -Input.GetAxis("Mouse X") * amout;
        float y = -Input.GetAxis("Mouse Y") * amout;
        //限制摆臂范围为两个最大值之间
        x = Mathf.Clamp(x, -maxAmout, maxAmout);
        y = Mathf.Clamp(y, -maxAmout, maxAmout);
        //手臂位置变化的值
        Vector3 finP = new Vector3(x, y, 0);
        //lerp是指两个点之间的线性插值
        transform.localPosition = Vector3.Lerp(transform.localPosition, originPosition + finP, Time.deltaTime * smoothAmout);
    }
}
